# fauXFOIL: an XFOIL-style command line and plot window for Scirocco.
#
# Builds against the Scirocco source tree next to this directory, or, with
# -DFAUXFOIL_SCIROCCO_SOURCE="" (or no such tree), against an installed
# Scirocco found with find_package.
cmake_minimum_required(VERSION 3.18)
project(fauXFOIL LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(GNUInstallDirs)

option(FAUXFOIL_GRAPHICS "Plot window with Qt 6 Widgets when Qt is found" ON)
option(FAUXFOIL_BUILD_TESTS "Build the fauXFOIL tests (needs GoogleTest)" ON)
set(FAUXFOIL_SCIROCCO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../scirocco"
    CACHE PATH "Scirocco source tree to build with; empty for an installed Scirocco")

# Scirocco
if(FAUXFOIL_SCIROCCO_SOURCE AND EXISTS "${FAUXFOIL_SCIROCCO_SOURCE}/CMakeLists.txt")
    # Only the library: its tests and examples belong to its own build
    set(SCIROCCO_BUILD_TESTS OFF)
    set(SCIROCCO_BUILD_EXAMPLES OFF)
    add_subdirectory("${FAUXFOIL_SCIROCCO_SOURCE}" scirocco EXCLUDE_FROM_ALL)
    get_directory_property(FAUXFOIL_SCIROCCO_VERSION
        DIRECTORY "${FAUXFOIL_SCIROCCO_SOURCE}" DEFINITION PROJECT_VERSION)
    add_library(scirocco::scirocco ALIAS scirocco)
    message(STATUS "fauXFOIL: Scirocco ${FAUXFOIL_SCIROCCO_VERSION} from ${FAUXFOIL_SCIROCCO_SOURCE}")
else()
    find_package(scirocco CONFIG REQUIRED)
    set(FAUXFOIL_SCIROCCO_VERSION ${scirocco_VERSION})
    message(STATUS "fauXFOIL: installed Scirocco ${FAUXFOIL_SCIROCCO_VERSION}")
endif()

# The plot font, compiled into the program: the .ttf becomes a byte array
set(FAUXFOIL_FONT "${CMAKE_CURRENT_SOURCE_DIR}/fonts/FauxfoilPlot.ttf")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FAUXFOIL_FONT}")
file(READ "${FAUXFOIL_FONT}" FAUXFOIL_FONT_HEX HEX)
file(SIZE "${FAUXFOIL_FONT}" FAUXFOIL_FONT_SIZE)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," FAUXFOIL_FONT_BYTES
       "${FAUXFOIL_FONT_HEX}")
string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],){16})" "\\1\n    "
       FAUXFOIL_FONT_BYTES "${FAUXFOIL_FONT_BYTES}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/font_data.cpp" CONTENT
"// Generated by CMake from fonts/FauxfoilPlot.ttf; do not edit.
#include \"font_data.h\"

namespace fauxfoil {

const unsigned char kPlotFontData[] = {
    @FAUXFOIL_FONT_BYTES@
};
const std::size_t kPlotFontSize = @FAUXFOIL_FONT_SIZE@;

}  // namespace fauxfoil
" @ONLY)

# Everything but the window and main()
add_library(fauxfoil_core STATIC
    ${CMAKE_CURRENT_BINARY_DIR}/font_data.cpp
    src/font.cpp
    src/airfoil.cpp
    src/bl_profiles.cpp
    src/format.cpp
    src/plot_scene.cpp
    src/plots.cpp
    src/polar.cpp
    src/postscript.cpp
    src/session.cpp
    src/solution.cpp
)
target_include_directories(fauxfoil_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(fauxfoil_core PUBLIC scirocco::scirocco)
target_compile_options(fauxfoil_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(fauXFOIL src/main.cpp)
target_compile_definitions(fauXFOIL PRIVATE
    SCIROCCO_VERSION="${FAUXFOIL_SCIROCCO_VERSION}")
install(TARGETS fauXFOIL RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

# Plot window: Qt 6 Widgets, optional; without it fauXFOIL is text-only
if(FAUXFOIL_GRAPHICS)
    find_package(Qt6 QUIET COMPONENTS Widgets)
endif()
if(FAUXFOIL_GRAPHICS AND Qt6Widgets_FOUND)
    message(STATUS "fauXFOIL: plot window with Qt ${Qt6Widgets_VERSION}")
    add_library(fauxfoil_qt STATIC src/qt/qt_display.cpp)
    target_link_libraries(fauxfoil_qt PUBLIC fauxfoil_core Qt6::Widgets)
    target_compile_options(fauxfoil_qt PRIVATE -Wall -Wextra)
    target_link_libraries(fauXFOIL PRIVATE fauxfoil_qt)
    target_compile_definitions(fauXFOIL PRIVATE FAUXFOIL_WITH_QT)
elseif(FAUXFOIL_GRAPHICS)
    message(STATUS "fauXFOIL: text only (Qt 6 Widgets not found)")
    target_link_libraries(fauXFOIL PRIVATE fauxfoil_core)
else()
    message(STATUS "fauXFOIL: text only (FAUXFOIL_GRAPHICS=OFF)")
    target_link_libraries(fauXFOIL PRIVATE fauxfoil_core)
endif()

# Tests
if(FAUXFOIL_BUILD_TESTS)
    find_package(GTest QUIET)
    if(GTest_FOUND)
        enable_testing()
        include(GoogleTest)
        # The banner (and so the plots and polar files) carries the
        # version CMake read from Scirocco
        add_test(NAME fauXFOIL.Version
            COMMAND ${CMAKE_COMMAND} -DFAUXFOIL=$<TARGET_FILE:fauXFOIL>
                -DEXPECTED=${FAUXFOIL_SCIROCCO_VERSION}
                -DWORK=${CMAKE_CURRENT_BINARY_DIR}
                -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/version_check.cmake)
        add_executable(fauxfoil_tests tests/fauxfoil_test.cpp)
        target_link_libraries(fauxfoil_tests PRIVATE fauxfoil_core GTest::gtest_main)
        gtest_discover_tests(fauxfoil_tests)
        if(TARGET fauxfoil_qt)
            add_executable(fauxfoil_qt_tests tests/fauxfoil_qt_test.cpp)
            target_link_libraries(fauxfoil_qt_tests PRIVATE fauxfoil_qt GTest::gtest)
            gtest_discover_tests(fauxfoil_qt_tests
                PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endif()
    else()
        message(STATUS "fauXFOIL: tests skipped (GoogleTest not found)")
    endif()
endif()
